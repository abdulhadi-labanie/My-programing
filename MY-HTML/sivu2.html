      function aloitaKello() {
        var aika = new Date();
        var h = aika.getHours();
        var m = aika.getMinutes();
        var s = aika.getSeconds();
        m = otaAika(m);
        s = otaAika(s);
        document.getElementById('txt').innerHTML =
        h + ":" + m + ":" + s;
        document.getElementById('txt_t').innerHTML =
        h + ":" + m + ":" + s;
        var t = setTimeout(aloitaKello, 500);
      }
    
      function otaAika(i) {
          if (i < 10) {i = "0" + i};  // add zero in front of numbers < 10
          return i;
      }
    </script>

      <button id="seiko" type="button" onclick="omaAika()">Näytä aika</button>

      <script>
      function omaAika() {
          setTimeout(function() {
              var aika = new Date();
              document.getElementById("seiko").innerHTML = aika.toLocaleTimeString();
              
              setInterval(function() {
                  var aika = new Date();
                  document.getElementById("seiko").innerHTML = aika.toLocaleTimeString();
              }, 2000); // Päivitä aika 2 sekunnin välein
          }, 3000); // Odota kolme sekuntia ennen kuin näytät ensimmäisen kerran
      }
      </script>
