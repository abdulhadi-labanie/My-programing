      <p id="kello_19"></p>
            
      <script>
          function updateClock() {
              var aika = new Date();
              var tunti = aika.getHours().toString();
              var minu = aika.getMinutes().toString().padStart(2, '0'); // Lisää nolla, jos minuutit ovat alle 10
              var kello = tunti + ":" + minu;
      
              document.getElementById("print_kello").innerHTML = tunti + ":" + minu;
      
              if (kello >= "5:01" && kello <= "10:00") {
                  document.getElementById("kello_19").innerHTML = "Hyvää huomenta 🌄";
              } else if (kello >= "10:01" && kello <= "14:00") {
                  document.getElementById("kello_19").innerHTML = "Hyvää päivää 🌞";
              } else if (kello >= "14:01" && kello <= "17:00") {
                  document.getElementById("kello_19").innerHTML = "Hyvää iltapäivää 🌅";
              } else if (kello >= "17:01" && kello <= "23:00") {
                  document.getElementById("kello_19").innerHTML = "Hyvää iltaa 🌚";
              } else {
                  document.getElementById("kello_19").innerHTML = "Hyvää yötä 🌛";
              }
          }
      
          updateClock();
          setInterval(updateClock, 60000);
          </script>
